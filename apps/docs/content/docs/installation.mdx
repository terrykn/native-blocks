## Getting Started

Run the `init` command to start a new project, or set up an existing one.

### CLI

<Tabs defaultValue="npm">
  <TabsList>
    <TabsTrigger value="npm">npm</TabsTrigger>
    <TabsTrigger value="bun">bun</TabsTrigger>
    <TabsTrigger value="pnpm">pnpm</TabsTrigger>
    <TabsTrigger value="yarn">yarn</TabsTrigger>
    <TabsTrigger value="yarn@berry">yarn @berry</TabsTrigger>
  </TabsList>
  <TabsContent value="npm">
    <BashCommand command={`npx native-blocks init`} />
  </TabsContent>
  <TabsContent value="bun">
    <BashCommand command={`bunx native-blocks init`} />
  </TabsContent>
  <TabsContent value="pnpm">
    <BashCommand command={`pnpm dlx native-blocks init`} />
  </TabsContent>
  <TabsContent value="yarn">
    <BashCommand command={`npx native-blocks init`} />
  </TabsContent>
  <TabsContent value="yarn@berry">
    <BashCommand command={`yarn dlx native-blocks init`} />
  </TabsContent>
</Tabs>

This command will initialize Native Blocks in your project and set up the necessary configurations.

### Add components

Once initialized, you can start adding components to your project.

<Tabs defaultValue="npm">
  <TabsList>
    <TabsTrigger value="npm">npm</TabsTrigger>
    <TabsTrigger value="bun">bun</TabsTrigger>
    <TabsTrigger value="pnpm">pnpm</TabsTrigger>
    <TabsTrigger value="yarn">yarn</TabsTrigger>
    <TabsTrigger value="yarn@berry">yarn @berry</TabsTrigger>
  </TabsList>
  <TabsContent value="npm">
    <BashCommand command={`npx native-blocks add button`} />
  </TabsContent>
  <TabsContent value="bun">
    <BashCommand command={`bunx native-blocks add button`} />
  </TabsContent>
  <TabsContent value="pnpm">
    <BashCommand command={`pnpm dlx native-blocks add button`} />
  </TabsContent>
  <TabsContent value="yarn">
    <BashCommand command={`npx native-blocks add button`} />
  </TabsContent>
  <TabsContent value="yarn@berry">
    <BashCommand command={`yarn dlx native-blocks add button`} />
  </TabsContent>
</Tabs>

The command above will add the `Button` component to your project. You can then import it like this:

```tsx title="app/index.tsx"
import { Button } from '@/components/ui/button';
import { Text } from '@/components/ui/text';

export default function Screen() {
  return (
    <Button>
      <Text>Click me</Text>
    </Button>
  );
}
```
