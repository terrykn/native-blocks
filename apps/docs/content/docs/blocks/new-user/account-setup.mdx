### Usage

```tsx
import { AccountSetup } from '@/components/blocks/account-setup';

export default function OnboardingScreen() {
  const handleFinalSubmit = (data: any) => {
    console.log('Onboarding data:', data);
  };

  return (
    <AccountSetup.Root totalSteps={3} onSubmit={handleFinalSubmit}>
      <AccountSetup.Header 
        title="Welcome" 
        description="Let's tailor your profile" 
      />
      
      <AccountSetup.Content>
        <AccountSetup.Step 
          index={0} 
          layoutType="grid" 
          options={OPTIONS_STEP_1} 
        />
        <AccountSetup.Step 
          index={1} 
          layoutType="list" 
          options={OPTIONS_STEP_2} 
        />
        <AccountSetup.Step 
          index={2} 
          layoutType="chips" 
          options={OPTIONS_STEP_3} 
          multiple 
        />
      </AccountSetup.Content>
      
      <AccountSetup.Footer />
    </AccountSetup.Root>
  );
}
```
